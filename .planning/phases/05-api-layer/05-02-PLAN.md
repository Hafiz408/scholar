import math, uuid, json
from datetime import datetime, timezone
from langchain_openai import ChatOpenAI
from app.models.schemas import StudyGoal, StudySession, StudyPlan
from pydantic import BaseModel

class GoalCreate(BaseModel):
    title: str
    topic: str
    knowledge_source_ids: list[str]
    deadline_days: int
    level: str  # beginner | intermediate | advanced
    sessions_per_week: int

async def generate_plan(goal_create: GoalCreate, goal_id: str) -> tuple[StudyGoal, list[StudySession]]:
    """Generate study plan. Returns (goal, sessions). Caller persists to SQLite."""
    total_sessions = math.ceil(goal_create.deadline_days / 7 * goal_create.sessions_per_week)
    total_sessions = max(total_sessions, 1)

    # Build prompt for gpt-4o-mini to generate session titles/topics
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.3)
    prompt = (
        f"You are a study planner. Generate exactly {total_sessions} study sessions for:\n"
        f"Topic: {goal_create.topic}\nLevel: {goal_create.level}\nDeadline: {goal_create.deadline_days} days\n\n"
        f"Return a JSON array of {total_sessions} objects, each with: "
        f'"title" (string), "topic" (string, sub-topic for the session), "estimated_minutes" (int 30-90).\n'
        f"Sequence sessions from foundational to advanced. No markdown, just the JSON array."
    )
    import asyncio
    response = await asyncio.to_thread(llm.invoke, prompt)
    # Parse JSON from response
    try:
        raw = response.content.strip()
        if raw.startswith("```"):
            raw = raw.split("```")[1].lstrip("json").strip()
        session_defs = json.loads(raw)
    except Exception:
        # Fallback: generate N generic sessions
        session_defs = [
            {"title": f"Session {i+1}: {goal_create.topic}", "topic": goal_create.topic, "estimated_minutes": 45}
            for i in range(total_sessions)
        ]

    now = datetime.now(timezone.utc).isoformat()
    goal = StudyGoal(
        id=goal_id, title=goal_create.title, topic=goal_create.topic,
        knowledge_source_ids=goal_create.knowledge_source_ids,
        deadline_days=goal_create.deadline_days, level=goal_create.level,
        sessions_per_week=goal_create.sessions_per_week,
        status="active", created_at=datetime.fromisoformat(now)
    )
    sessions = [
        StudySession(
            id=str(uuid.uuid4()), goal_id=goal_id, session_number=i+1,
            title=s.get("title", f"Session {i+1}"),
            topic=s.get("topic", goal_create.topic),
            estimated_minutes=s.get("estimated_minutes", 45),
            status="pending", created_at=datetime.fromisoformat(now)
        )
        for i, s in enumerate(session_defs[:total_sessions])
    ]
    return goal, sessions
```

import asyncio, json
from collections.abc import AsyncIterable
from langchain_openai import ChatOpenAI
from app.models.schemas import RetrievedChunk

async def stream_notes(
    session_id: str,
    session_topic: str,
    level: str,
    context_chunks: list[RetrievedChunk]
) -> AsyncIterable[str]:
    """Async generator: yields plain text chunks of notes as they stream from LLM."""
    context_text = "\n\n".join(
        f"[{c.source_title}, p.{c.page_number}] {c.content}"
        for c in context_chunks
    )
    prompt = (
        f"You are a study note generator. Generate comprehensive study notes for a {level} student.\n"
        f"Topic: {session_topic}\n\n"
        f"Use ONLY the following source material. For every factual claim, cite the source as "
        f"(Book Title, p. N). Never use training knowledge.\n\n"
        f"Source material:\n{context_text}\n\n"
        f"Write in clear, structured markdown with headers and bullet points."
    )
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.2, streaming=True)

    # Use asyncio.to_thread for sync streaming (per Phase 3 decision)
    collected = []
    def _stream():
        for chunk in llm.stream(prompt):
            if chunk.content:
                collected.append(chunk.content)
        return collected

    # For async streaming, use a queue-based approach
    import queue as q_module
    chunk_queue: asyncio.Queue = asyncio.Queue()

    async def _fill_queue():
        loop = asyncio.get_event_loop()
        chunks = await asyncio.to_thread(_stream)
        for c in chunks:
            await chunk_queue.put(c)
        await chunk_queue.put(None)  # sentinel

    asyncio.create_task(_fill_queue())

    while True:
        chunk = await chunk_queue.get()
        if chunk is None:
            break
        yield chunk

**session_chat.py — stream_chat:**

```python
import json, asyncio
from collections.abc import AsyncIterable
from datetime import datetime, timezone
import uuid
import aiosqlite
from langchain_openai import ChatOpenAI
from langchain.schema import HumanMessage, SystemMessage, AIMessage
from fastapi.sse import ServerSentEvent
from app.models.schemas import RetrievedChunk

async def stream_chat(
    message: str,
    session_id: str,
    context_chunks: list[RetrievedChunk],
    db: aiosqlite.Connection
) -> AsyncIterable[ServerSentEvent]:
    """Async generator: yields ServerSentEvent objects for token streaming."""
    # 1. Load existing chat history from SQLite
    async with db.execute(
        "SELECT role, content FROM chat_history WHERE session_id=? ORDER BY rowid ASC",
        (session_id,)
    ) as cur:
        history_rows = await cur.fetchall()

    context_text = "\n\n".join(
        f"[{c.source_title}, p.{c.page_number}] {c.content}"
        for c in context_chunks
    )
    system = SystemMessage(content=(
        "You are a study assistant. Answer ONLY from the provided context. "
        "If the context does not contain the answer, say: 'I cannot find that in your study materials.' "
        "Cite sources as (Book Title, p. N) after each claim.\n\n"
        f"Context:\n{context_text}"
    ))
    messages = [system]
    for row in history_rows:
        if row["role"] == "user":
            messages.append(HumanMessage(content=row["content"]))
        else:
            messages.append(AIMessage(content=row["content"]))
    messages.append(HumanMessage(content=message))

    # 2. Persist user message
    now = datetime.now(timezone.utc).isoformat()
    await db.execute(
        "INSERT INTO chat_history (id, session_id, role, content, citations, created_at) VALUES (?,?,?,?,?,?)",
        (str(uuid.uuid4()), session_id, "user", message, "[]", now)
    )
    await db.commit()

    # 3. Stream tokens using asyncio.to_thread (per Phase 3 decision)
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.2, streaming=True)
    full_response = []

    chunk_queue: asyncio.Queue = asyncio.Queue()

    def _stream_sync():
        for chunk in llm.stream(messages):
            if chunk.content:
                full_response.append(chunk.content)
                import asyncio as aio
                aio.get_event_loop().call_soon_threadsafe(chunk_queue.put_nowait, chunk.content)
        aio.get_event_loop().call_soon_threadsafe(chunk_queue.put_nowait, None)

    asyncio.create_task(asyncio.to_thread(_stream_sync))

    while True:
        token = await chunk_queue.get()
        if token is None:
            break
        yield ServerSentEvent(
            data=json.dumps({"type": "token", "content": token}),
            event="token"
        )

    # 4. Persist assistant response
    assistant_content = "".join(full_response)
    citations_json = json.dumps([c.dict() for c in context_chunks[:3]])
    await db.execute(
        "INSERT INTO chat_history (id, session_id, role, content, citations, created_at) VALUES (?,?,?,?,?,?)",
        (str(uuid.uuid4()), session_id, "assistant", assistant_content, citations_json, now)
    )
    await db.commit()

    # 5. Emit citations then done
    yield ServerSentEvent(
        data=json.dumps({"type": "citations", "chunks": [c.dict() for c in context_chunks[:3]]}),
        event="citations"
    )
    yield ServerSentEvent(data=json.dumps({"type": "done"}), event="done")
```

Implementation note: The streaming approach above uses asyncio.Queue with call_soon_threadsafe to bridge sync LangChain streaming into async context — matching Phase 3's established pattern for asyncio.to_thread + sync LangChain. If session_chat imports fail due to missing langchain.schema, use from langchain_core.messages import HumanMessage, SystemMessage, AIMessage instead.

import json, uuid
from datetime import datetime, timezone
from fastapi import APIRouter, Depends, HTTPException
import aiosqlite
from app.db.database import get_db
from app.models.schemas import StudyGoal, StudySession, StudyPlan
from app.agents.planner import generate_plan, GoalCreate

router = APIRouter()

@router.post("", response_model=StudyPlan, status_code=201)
async def create_goal(goal_in: GoalCreate, db: aiosqlite.Connection = Depends(get_db)):
    # 1. Validate knowledge_source_ids exist
    for src_id in goal_in.knowledge_source_ids:
        async with db.execute("SELECT id FROM knowledge_sources WHERE id=?", (src_id,)) as cur:
            if not await cur.fetchone():
                raise HTTPException(status_code=422, detail=f"Knowledge source {src_id} not found")

    # 2. Generate plan via planner agent
    goal_id = str(uuid.uuid4())
    goal, sessions = await generate_plan(goal_in, goal_id)

    # 3. Persist goal
    await db.execute(
        "INSERT INTO study_goals (id, title, topic, knowledge_source_ids, deadline_days, level, sessions_per_week, status, created_at) "
        "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
        (goal.id, goal.title, goal.topic, json.dumps(goal.knowledge_source_ids),
         goal.deadline_days, goal.level, goal.sessions_per_week, goal.status,
         goal.created_at.isoformat())
    )

    # 4. Persist sessions
    for s in sessions:
        await db.execute(
            "INSERT INTO study_sessions (id, goal_id, session_number, title, topic, estimated_minutes, status, quiz_score, notes_markdown, created_at) "
            "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
            (s.id, s.goal_id, s.session_number, s.title, s.topic,
             s.estimated_minutes, s.status, None, None, s.created_at.isoformat())
        )
    await db.commit()

    return StudyPlan(
        goal=goal, sessions=sessions,
        total_sessions=len(sessions),
        completed_sessions=0
    )

import json
from collections.abc import AsyncIterable
from fastapi import APIRouter, Depends, HTTPException
from fastapi.sse import EventSourceResponse, ServerSentEvent
import aiosqlite
from app.db.database import get_db
from app.agents.note_generator import stream_notes
from app.retrieval.hybrid_retriever import HybridRetriever

router = APIRouter()

@router.post("/{session_id}/start", response_class=EventSourceResponse)
async def start_session(session_id: str, db: aiosqlite.Connection = Depends(get_db)):
    # 1. Load session
    async with db.execute("SELECT * FROM study_sessions WHERE id=?", (session_id,)) as cur:
        row = await cur.fetchone()
    if not row:
        raise HTTPException(status_code=404, detail="Session not found")
    if row["status"] in ("in_progress", "complete"):
        raise HTTPException(status_code=409, detail=f"Session already {row['status']}")

    # 2. Load goal for context
    async with db.execute("SELECT * FROM study_goals WHERE id=?", (row["goal_id"],)) as cur:
        goal_row = await cur.fetchone()
    source_ids = json.loads(goal_row["knowledge_source_ids"])

    # 3. Mark in_progress
    await db.execute("UPDATE study_sessions SET status='in_progress' WHERE id=?", (session_id,))
    await db.commit()

    # 4. Retrieve context then stream notes
    async def _generate() -> AsyncIterable[ServerSentEvent]:
        retriever = HybridRetriever()
        result = await retriever.retrieve(row["topic"], source_ids)
        full_notes = []
        async for chunk in stream_notes(session_id, row["topic"], goal_row["level"], result.chunks):
            full_notes.append(chunk)
            yield ServerSentEvent(
                data=json.dumps({"type": "notes_chunk", "content": chunk}),
                event="notes_chunk"
            )
        # 5. Persist complete notes
        notes_md = "".join(full_notes)
        import aiosqlite as _aio
        async with _aio.connect(db._path if hasattr(db, '_path') else __import__('app.config', fromlist=['settings']).settings.sqlite_path) as persist_db:
            persist_db.row_factory = _aio.Row
            await persist_db.execute(
                "UPDATE study_sessions SET notes_markdown=? WHERE id=?",
                (notes_md, session_id)
            )
            await persist_db.commit()
        total = len(notes_md)
        yield ServerSentEvent(
            data=json.dumps({"type": "notes_done", "total_chars": total}),
            event="notes_done"
        )

    return EventSourceResponse(_generate())
```

Inside the SSE generator, the response has already been initiated so the outer `db` connection may be closed by the time notes finish streaming. Use `settings.sqlite_path` directly in a new connection for the final persist. Import: `from app.config import settings`.

import json
from collections.abc import AsyncIterable
from fastapi import APIRouter, Depends, HTTPException
from fastapi.sse import EventSourceResponse, ServerSentEvent
import aiosqlite
from pydantic import BaseModel
from app.db.database import get_db
from app.agents.session_chat import stream_chat
from app.retrieval.hybrid_retriever import HybridRetriever

class ChatRequest(BaseModel):
    session_id: str
    message: str

router = APIRouter()

@router.post("/stream", response_class=EventSourceResponse)
async def stream_chat_endpoint(body: ChatRequest, db: aiosqlite.Connection = Depends(get_db)):
    async with db.execute("SELECT * FROM study_sessions WHERE id=?", (body.session_id,)) as cur:
        row = await cur.fetchone()
    if not row:
        raise HTTPException(status_code=404, detail="Session not found")

    async with db.execute("SELECT * FROM study_goals WHERE id=?", (row["goal_id"],)) as cur:
        goal_row = await cur.fetchone()
    source_ids = json.loads(goal_row["knowledge_source_ids"])

    async def _stream() -> AsyncIterable[ServerSentEvent]:
        retriever = HybridRetriever()
        result = await retriever.retrieve(body.message, source_ids)
        async for event in stream_chat(body.message, body.session_id, result.chunks, db):
            yield event

    return EventSourceResponse(_stream())
