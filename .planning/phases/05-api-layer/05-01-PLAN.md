**routers/quiz.py — implement both endpoints:**

```python
import json, uuid
from datetime import datetime, timezone
from fastapi import APIRouter, Depends, HTTPException, Query
import aiosqlite
from app.db.database import get_db
from app.models.schemas import QuizQuestion, QuizSubmission, QuizResult
from app.agents.quiz_agent import generate_quiz, evaluate_quiz
from pydantic import BaseModel

class QuizQuestionPublic(BaseModel):
    """Safe quiz question — correct_index and explanation excluded (revealed post-submit)."""
    id: str
    question: str
    options: list[str]

router = APIRouter()

@router.post("/generate", response_model=list[QuizQuestionPublic])
async def generate_quiz_endpoint(
    session_id: str = Query(..., description="Session ID to generate quiz for"),
    db: aiosqlite.Connection = Depends(get_db)
):
    # 1. Validate session exists and has status 'in_progress' or 'complete'
    async with db.execute("SELECT id, status FROM study_sessions WHERE id=?", (session_id,)) as cur:
        row = await cur.fetchone()
    if not row:
        raise HTTPException(status_code=404, detail="Session not found")

    # 2. Call quiz agent to generate questions
    questions: list[QuizQuestion] = await generate_quiz(session_id, db)

    # 3. Store full questions (with correct_index) in quiz_questions table
    now = datetime.now(timezone.utc).isoformat()
    for q in questions:
        await db.execute(
            "INSERT OR REPLACE INTO quiz_questions "
            "(id, session_id, question, options, correct_index, explanation, created_at) "
            "VALUES (?, ?, ?, ?, ?, ?, ?)",
            (q.id, session_id, q.question, json.dumps(q.options), q.correct_index, q.explanation, now)
        )
    await db.commit()

    # 4. Return stripped version (no correct_index, no explanation)
    return [QuizQuestionPublic(id=q.id, question=q.question, options=q.options) for q in questions]


@router.post("/submit", response_model=QuizResult)
async def submit_quiz(
    submission: QuizSubmission,
    db: aiosqlite.Connection = Depends(get_db)
):
    # 1. Load stored questions from SQLite
    async with db.execute(
        "SELECT id, question, options, correct_index, explanation FROM quiz_questions WHERE session_id=?",
        (submission.session_id,)
    ) as cur:
        rows = await cur.fetchall()
    if not rows:
        raise HTTPException(status_code=404, detail="No quiz found for this session. Call /quiz/generate first.")

    questions = [
        QuizQuestion(
            id=r["id"], question=r["question"],
            options=json.loads(r["options"]),
            correct_index=r["correct_index"],
            explanation=r["explanation"]
        )
        for r in rows
    ]

    # 2. Evaluate answers (pure function — no LLM call)
    result: QuizResult = evaluate_quiz(questions, submission.answers)

    # 3. Update session: quiz_score + status='complete'
    await db.execute(
        "UPDATE study_sessions SET quiz_score=?, status='complete' WHERE id=?",
        (result.score, submission.session_id)
    )

    # 4. Check if all sessions in goal are complete → update goal status
    async with db.execute(
        "SELECT goal_id FROM study_sessions WHERE id=?", (submission.session_id,)
    ) as cur:
        session_row = await cur.fetchone()
    if session_row:
        goal_id = session_row["goal_id"]
        async with db.execute(
            "SELECT COUNT(*) as total, SUM(CASE WHEN status='complete' THEN 1 ELSE 0 END) as done "
            "FROM study_sessions WHERE goal_id=?", (goal_id,)
        ) as cur:
            counts = await cur.fetchone()
        if counts and counts["total"] == counts["done"]:
            await db.execute("UPDATE study_goals SET status='complete' WHERE id=?", (goal_id,))

    await db.commit()
    return result
```
